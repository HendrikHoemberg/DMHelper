```java
package dev.hendrikhoemberg.dmhelper.common.config;

import jakarta.servlet.http.Cookie;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.web.servlet.HandlerInterceptor;

import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

public class PinInterceptor implements HandlerInterceptor {

    private static final Logger log = LoggerFactory.getLogger(PinInterceptor.class);
    private static final int MAX_FAILURES_BEFORE_DELAY = 5;
    private static final int MAX_FAILURES_BEFORE_BLOCK = 10;
    private static final long BLOCK_DURATION_MS = 30_000;
    private static final long DELAY_MS = 2_000;

    private final PinManager pinManager;
    private final Map<String, RateLimitEntry> rateLimit = new ConcurrentHashMap<>();

    private record RateLimitEntry(int failures, long firstFailureTime, long blockedUntil) {}

    public PinInterceptor(PinManager pinManager) {
        this.pinManager = pinManager;
    }

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response,
                             Object handler) throws Exception {
        String pinCookie = extractPinCookie(request);
        if (pinManager.isValid(pinCookie)) {
            rateLimit.remove(request.getRemoteAddr());
            return true;
        }

        String ip = request.getRemoteAddr();
        RateLimitEntry entry = rateLimit.compute(ip, (k, v) -> {
            long now = System.currentTimeMillis();
            if (v == null) return new RateLimitEntry(1, now, 0);
            if (now < v.blockedUntil()) return v;
            return new RateLimitEntry(v.failures() + 1, v.firstFailureTime(), 0);
        });

        if (entry.failures() >= MAX_FAILURES_BEFORE_BLOCK) {
            long now = System.currentTimeMillis();
            long blockedUntil = now + BLOCK_DURATION_MS;
            rateLimit.put(ip, new RateLimitEntry(entry.failures(), entry.firstFailureTime(), blockedUntil));
            response.sendError(429, "Too many PIN attempts");
            return false;
        }

        if (entry.failures() >= MAX_FAILURES_BEFORE_DELAY) {
            try {
                Thread.sleep(DELAY_MS);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }

        response.setContentType("text/html;charset=UTF-8");
        response.getWriter().write("""
                <!DOCTYPE html>
                <html lang="en">
                <head><meta charset="UTF-8"><title>DM Helper — PIN Required</title>
                <style>body{font-family:sans-serif;display:flex;justify-content:center;align-items:center;height:100vh;margin:0;background:#1a1a2e;color:#eee}form{background:#16213e;padding:2rem;border-radius:8px;text-align:center}input{font-size:1.5rem;text-align:center;letter-spacing:0.5em;padding:0.5rem;margin:1rem 0;border:1px solid #555;border-radius:4px;background:#0f3460;color:#fff}button{font-size:1rem;padding:0.5rem 2rem;background:#e94560;color:#fff;border:none;border-radius:4px;cursor:pointer}</style></head>
                <body><form onsubmit="event.preventDefault();document.cookie='dm_pin='+document.getElementById('pin').value+';path=/;SameSite=Lax';location.reload()"><h1>DM Helper</h1><p>Enter the PIN shown in the terminal:</p><input id="pin" type="text" maxlength="6" autocomplete="off" autofocus><br><button type="submit">Unlock</button></form></body></html>""");
        return false;
    }

    private String extractPinCookie(HttpServletRequest request) {
        Cookie[] cookies = request.getCookies();
        if (cookies != null) {
            for (Cookie cookie : cookies) {
                if ("dm_pin".equals(cookie.getName())) {
                    return cookie.getValue();
                }
            }
        }
        return null;
    }
}
```
